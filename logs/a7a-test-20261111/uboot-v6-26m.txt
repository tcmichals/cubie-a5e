# U-Boot commands, sent one per line by bootcap.py --uboot (also usable by hand)
mw.l 0x020025a4 0x00030001
mw.l 0x02003340 0x80010001
mw.l 0x0200335c 0x00010001
setenv bootargs "console=ttyS0,115200 root=/dev/mmcblk0p2 rootwait rw panic=10 loglevel=8 clk_ignore_unused fw_devlink=permissive"
load mmc 0:1 0x4fa00000 sun60i-a733-cubie-a7a-26m.dtb
fdt addr 0x4fa00000
fdt resize 0x10000
load mmc 0:1 0x4fe00000 cubie-a7a-rproc.dtbo
fdt apply 0x4fe00000
load mmc 0:1 0x4fe00000 cubie-a7a-ov5647.dtbo
fdt apply 0x4fe00000
load mmc 0:1 0x40200000 Image
booti 0x40200000 - 0x4fa00000
